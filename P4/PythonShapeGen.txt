import random

def random_float(min_val, max_val):
    return round(random.uniform(min_val, max_val), 2)

def random_color():
    r = random_float(0.1, 1.0)
    g = random_float(0.1, 1.0)
    b = random_float(0.1, 1.0)
    return r, g, b

def generate_scene(num_shapes=10):
    print("scene 0")
    print("# type x y z width height depth r g b a rotx roty rotz radius segments")
    
    for _ in range(num_shapes):
        shape_type = random.choice([1, 3, 5, 6])   # 1=cuboid, 3=cone?, 5/6=cylinder variants
        
        # Position: limited to [-50, 50]
        x = random_float(-50, 50)
        y = random_float(-50, 50)
        z = random_float(-50, 50)
        
        # Size parameters: width, height, depth, radius limited to [1, 8]
        width  = random_float(1, 8)
        height = random_float(1, 8)
        depth  = random_float(1, 8)
        radius = random_float(1, 8)
        
        # Random bright-ish color with alpha = 1.0
        r, g, b = random_color()
        a = 1.0
        
        # Rotation angles (in radians, small random values)
        rotx = random_float(0, 3.14)
        roty = random_float(0, 3.14)
        rotz = random_float(0, 3.14)
        
        # Segments for round shapes (only used by cone/cylinder)
        segments = random.randint(2, 256)
        
        # Print in the exact requested format
        print(f"shape {shape_type} {x} {y} {z} {width} {height} {depth} "
              f"{r} {g} {b} {a} {rotx} {roty} {rotz} {radius} {segments}")

# ======================
# Run the generator
# ======================

print("=== Random 3D Scene Generator ===\n")
generate_scene(num_shapes=1000)   # Change this number if you want more/fewer shapes
