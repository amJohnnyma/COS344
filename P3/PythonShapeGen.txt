

import random

def generate_test_scene(
    num_shapes=100,
    x_min=-9.0, x_max=9.0,
    y_min=-9.0, y_max=9.0,
    z_min=-3.0, z_max=3.0,
    width_min=2.0, width_max=6.0,
    height_min=2.0, height_max=6.0,
    depth_min=1.5, depth_max=4.5,
    filename="test.txt"
):
    print(f"Generating scene with {num_shapes} triangular prisms...")
    

        
for i in range(num_shapes):
    x = round(random.uniform(x_min, x_max), 4)
    y = round(random.uniform(y_min, y_max), 4)
    z = round(random.uniform(z_min, z_max), 4)
    
    width  = round(random.uniform(width_min, width_max), 4)
    height = round(random.uniform(height_min, height_max), 4)
    depth  = round(random.uniform(depth_min, depth_max), 4)
    
    # Nice varied colors
    r = round(random.uniform(0.1, 1.0), 4)
    g = round(random.uniform(0.1, 1.0), 4)
    b = round(random.uniform(0.1, 1.0), 4)
    a = 1.0
    
    # Occasional rotation (most shapes flat, some tilted)
    rotx = round(random.uniform(-0.8, 0.8), 4) if random.random() < 0.35 else 0.0
    roty = round(random.uniform(-0.8, 0.8), 4) if random.random() < 0.30 else 0.0
    rotz = round(random.uniform(-1.2, 1.2), 4) if random.random() < 0.25 else 0.0
    
    line = f"shape 3 {x} {y} {z} {width} {height} {depth} {r} {g} {b} {a} {rotx} {roty} {rotz}\n"
    print(line)
    


if __name__ == "__main__":
    generate_test_scene(
        num_shapes=150,           # Change this to 50, 100, 200, 300, 500 etc.
        x_min=-9.5,
        x_max=9.5,
        y_min=-9.5,
        y_max=9.5,
        z_min=-4.0,
        z_max=4.0,
        width_min=1.8,
        width_max=5.5,
        height_min=1.8,
        height_max=5.5,
        depth_min=1.2,
        depth_max=4.8
    )
